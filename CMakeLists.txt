cmake_minimum_required(VERSION 3.16)
project(g1_deploy LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(CONTROL_SOURCE "${REPOSITORY_ROOT}/include")
set(ONNXRUNTIME_ROOT "$ENV{ONNXRUNTIME_ROOT}" CACHE PATH "Native CPU ONNX Runtime C/C++ distribution")
if(NOT ONNXRUNTIME_ROOT)
    message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to the ONNX Runtime distribution matching this onboard CPU")
endif()
find_package(Threads REQUIRED)
list(APPEND CMAKE_PREFIX_PATH "/usr/local" "/opt/unitree_robotics")
find_package(unitree_sdk2 REQUIRED)
get_target_property(SDK_LIBRARY unitree_sdk2 IMPORTED_LOCATION)
if(NOT SDK_LIBRARY)
    get_target_property(SDK_LIBRARY unitree_sdk2 IMPORTED_LOCATION_RELEASE)
endif()
if(NOT SDK_LIBRARY)
    message(FATAL_ERROR "The installed unitree_sdk2 package must expose its native library location")
endif()
get_filename_component(SDK_RUNTIME_DIR "${SDK_LIBRARY}" DIRECTORY)
find_package(Boost REQUIRED)
find_package(yaml-cpp REQUIRED)
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH)
find_library(ORT_LIBRARY NAMES onnxruntime libonnxruntime.so.1 PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH)
if(NOT ORT_INCLUDE OR NOT ORT_LIBRARY)
    message(FATAL_ERROR "ONNXRUNTIME_ROOT must contain include/onnxruntime_cxx_api.h and lib/libonnxruntime.so")
endif()

add_executable(g1_dwaq "${REPOSITORY_ROOT}/src/main.cpp")
target_include_directories(g1_dwaq PRIVATE "${CONTROL_SOURCE}" "${ORT_INCLUDE}")
target_link_libraries(g1_dwaq PRIVATE unitree_sdk2 Boost::boost yaml-cpp "${ORT_LIBRARY}")
target_compile_definitions(g1_dwaq PRIVATE G1_WITH_POLICY
    G1_POLICY_DEFAULT_CONFIG="${REPOSITORY_ROOT}/config/g1_locomotion.yaml"
    G1_POLICY_DEFAULT_MODEL="${REPOSITORY_ROOT}/models/policy.onnx")
# Launcher passes package-local resources. Keep ORT local after installation.
set_target_properties(g1_dwaq PROPERTIES INSTALL_RPATH "$ORIGIN/../lib;${SDK_RUNTIME_DIR}")
if(UNIX AND NOT APPLE)
    target_link_options(g1_dwaq PRIVATE "-Wl,--disable-new-dtags")
endif()

include(CTest)
if(BUILD_TESTING)
    foreach(test IN ITEMS g1_hardware g1_gamepad g1_unitree_remote g1_locomotion_policy)
        add_executable(${test}_test "${REPOSITORY_ROOT}/tests/${test}_test.cpp")
        target_include_directories(${test}_test PRIVATE "${CONTROL_SOURCE}" "${ORT_INCLUDE}")
    endforeach()
    target_link_libraries(g1_hardware_test PRIVATE unitree_sdk2 yaml-cpp)
    target_link_libraries(g1_locomotion_policy_test PRIVATE yaml-cpp "${ORT_LIBRARY}")
    add_test(NAME g1_hardware COMMAND g1_hardware_test "${REPOSITORY_ROOT}/config/g1_locomotion.yaml")
    add_test(NAME g1_gamepad COMMAND g1_gamepad_test)
    add_test(NAME g1_unitree_remote COMMAND g1_unitree_remote_test)
    add_test(NAME g1_locomotion_policy COMMAND g1_locomotion_policy_test
        "${REPOSITORY_ROOT}/models/policy.onnx" "${REPOSITORY_ROOT}/config/g1_locomotion.yaml")
endif()

install(TARGETS g1_dwaq RUNTIME DESTINATION bin)
install(FILES "${REPOSITORY_ROOT}/models/policy.onnx" DESTINATION models)
install(FILES "${REPOSITORY_ROOT}/config/g1_locomotion.yaml" DESTINATION config)
# Preserve SONAME symlinks and their targets from the selected native ORT release.
install(DIRECTORY "${ONNXRUNTIME_ROOT}/lib/" DESTINATION lib
    FILES_MATCHING PATTERN "libonnxruntime.so*" PATTERN "libonnxruntime_providers_shared.so*")
foreach(notice IN ITEMS LICENSE ThirdPartyNotices.txt Privacy.md VERSION_NUMBER)
    install(FILES "${ONNXRUNTIME_ROOT}/${notice}" DESTINATION licenses/onnxruntime OPTIONAL)
endforeach()
install(PROGRAMS "${REPOSITORY_ROOT}/scripts/g1.sh" DESTINATION . RENAME g1)
install(PROGRAMS "${REPOSITORY_ROOT}/scripts/monitor.sh" DESTINATION . RENAME monitor)
install(FILES "${REPOSITORY_ROOT}/web/joint_monitor.py" "${REPOSITORY_ROOT}/web/requirements.txt" DESTINATION web)
install(FILES "${REPOSITORY_ROOT}/web/static/joint_monitor.html" DESTINATION web/static)
install(FILES "${REPOSITORY_ROOT}/README.md" "${REPOSITORY_ROOT}/ORIGIN.md" "${REPOSITORY_ROOT}/CHANGELOG.md" DESTINATION .)
install(FILES "${REPOSITORY_ROOT}/docs/HARDWARE.md" DESTINATION docs)
install(FILES "${REPOSITORY_ROOT}/docs/WEB.md" DESTINATION docs)
install(FILES "${REPOSITORY_ROOT}/LICENSE" DESTINATION licenses/project)
configure_file("${REPOSITORY_ROOT}/cmake/BUILD_INFO.in" "${CMAKE_CURRENT_BINARY_DIR}/BUILD_INFO.txt" @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/BUILD_INFO.txt" DESTINATION .)
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_NAME "g1-deploy")
set(CPACK_PACKAGE_VERSION "1.0.0")
set(CPACK_PACKAGE_FILE_NAME "g1-deploy-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
